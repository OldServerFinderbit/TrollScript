-- // UTILS
local function cleanName(n)
    return n:gsub("%s+", "")
end

local function getChar(plr)
    return plr and plr.Character
end

local function getHum(plr)
    local c = getChar(plr)
    return c and c:FindFirstChildOfClass("Humanoid")
end

local function getRoot(plr)
    local c = getChar(plr)
    return c and (c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("Torso") or c.PrimaryPart)
end

local function notify(msg, color)
    color = color or THEME.accent
    local sg = Instance.new("ScreenGui")
    sg.Name = "TrollNotify"
    sg.ResetOnSpawn = false
    sg.Parent = LocalPlayer:WaitForChild("PlayerGui")
    local f = Instance.new("Frame")
    f.Size = UDim2.new(0, 260, 0, 36)
    f.Position = UDim2.new(0.5, -130, 0, -50)
    f.BackgroundColor3 = THEME.panel
    f.BorderSizePixel = 0
    f.Parent = sg
    Instance.new("UICorner", f).CornerRadius = UDim.new(0, 8)
    local t = Instance.new("TextLabel")
    t.Size = UDim2.new(1, -16, 1, 0)
    t.Position = UDim2.new(0, 8, 0, 0)
    t.BackgroundTransparency = 1
    t.Text = msg
    t.TextColor3 = color
    t.Font = Enum.Font.GothamMedium
    t.TextSize = 13
    t.TextXAlignment = Enum.TextXAlignment.Left
    t.Parent = f
    TweenService:Create(f, TweenInfo.new(0.25), {Position = UDim2.new(0.5, -130, 0, 18)}):Play()
    task.delay(2.2, function()
        TweenService:Create(f, TweenInfo.new(0.2), {Position = UDim2.new(0.5, -130, 0, -50)}):Play()
        task.wait(0.25)
        sg:Destroy()
    end)
end

local function makeBtn(name, order, color)
    local b = Instance.new("TextButton")
    b.Name = name
    b.LayoutOrder = order
    b.BackgroundColor3 = color or THEME.panel
    b.Text = name
    b.TextColor3 = THEME.text
    b.Font = Enum.Font.GothamMedium
    b.TextSize = 12
    b.BorderSizePixel = 0
    b.AutoButtonColor = false
    b.Parent = Actions
    Instance.new("UICorner", b).CornerRadius = UDim.new(0, 7)
    b.MouseEnter:Connect(function()
        TweenService:Create(b, TweenInfo.new(0.12), {BackgroundColor3 = THEME.accent}):Play()
    end)
    b.MouseLeave:Connect(function()
        TweenService:Create(b, TweenInfo.new(0.12), {BackgroundColor3 = color or THEME.panel}):Play()
    end)
    return b
end

-- // PLAYER LIST LOGIC
local function clearList()
    for _, c in ipairs(ListFrame:GetChildren()) do
        if c:IsA("TextButton") then c:Destroy() end
    end
end

local function refreshPlayers()
    clearList()
    local y = 0
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr \~= LocalPlayer then
            local btn = Instance.new("TextButton")
            btn.Size = UDim2.new(1, -4, 0, 28)
            btn.BackgroundColor3 = Color3.fromRGB(36, 36, 44)
            btn.Text = "  " .. plr.Name
            btn.TextColor3 = THEME.text
            btn.Font = Enum.Font.Gotham
            btn.TextSize = 12
            btn.TextXAlignment = Enum.TextXAlignment.Left
            btn.BorderSizePixel = 0
            btn.AutoButtonColor = false
            btn.Parent = ListFrame
            Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 5)

            btn.MouseButton1Click:Connect(function()
                selectedPlayer = plr
                SelectedLbl.Text = "target: " .. plr.Name
                SelectedLbl.TextColor3 = THEME.accent
                notify("selected " .. plr.Name)
            end)
            btn.MouseEnter:Connect(function()
                TweenService:Create(btn, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(50, 50, 62)}):Play()
            end)
            btn.MouseLeave:Connect(function()
                TweenService:Create(btn, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(36, 36, 44)}):Play()
            end)
            y = y + 32
        end
    end
    ListFrame.CanvasSize = UDim2.new(0, 0, 0, math.max(y + 8, 140))
end

-- // ACTIONS (FE-safe methods)
local function doKill()
    if not selectedPlayer then return notify("no target", THEME.danger) end
    local hum = getHum(selectedPlayer)
    if not hum then return notify("no character", THEME.danger) end
    -- classic FE kill via health + break joints if possible
    pcall(function()
        hum.Health = 0
        local char = getChar(selectedPlayer)
        if char then
            for _, p in ipairs(char:GetDescendants()) do
                if p:IsA("BasePart") then
                    p.CanCollide = false
                    p.Velocity = Vector3.new(0, -50, 0)
                end
            end
        end
    end)
    notify("killed " .. selectedPlayer.Name, THEME.success)
end

local function doFling()
    if not selectedPlayer then return notify("no target", THEME.danger) end
    local root = getRoot(selectedPlayer)
    local myRoot = getRoot(LocalPlayer)
    if not root or not myRoot then return notify("missing root", THEME.danger) end
    pcall(function()
        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)
        bv.Velocity = Vector3.new(9999, 9999, 9999)
        bv.Parent = root
        task.delay(0.15, function()
            if bv and bv.Parent then bv:Destroy() end
        end)
        -- also spin
        local bang = Instance.new("BodyAngularVelocity")
        bang.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
        bang.AngularVelocity = Vector3.new(0, 9999, 0)
        bang.Parent = root
        task.delay(0.2, function()
            if bang and bang.Parent then bang:Destroy() end
        end)
    end)
    notify("flung " .. selectedPlayer.Name, THEME.success)
end

local function doBring()
    if not selectedPlayer then return notify("no target", THEME.danger) end
    local root = getRoot(selectedPlayer)
    local myRoot = getRoot(LocalPlayer)
    if not root or not myRoot then return notify("missing root", THEME.danger) end
    pcall(function()
        root.CFrame = myRoot.CFrame * CFrame.new(0, 0, -3)
    end)
    notify("brought " .. selectedPlayer.Name, THEME.success)
end

local function doGoto()
    if not selectedPlayer then return notify("no target", THEME.danger) end
    local root = getRoot(selectedPlayer)
    local myRoot = getRoot(LocalPlayer)
    if not root or not myRoot then return notify("missing root", THEME.danger) end
    pcall(function()
        myRoot.CFrame = root.CFrame * CFrame.new(0, 0, 3)
    end)
    notify("went to " .. selectedPlayer.Name, THEME.success)
end

local function doFreeze()
    if not selectedPlayer then return notify("no target", THEME.danger) end
    local hum = getHum(selectedPlayer)
    local root = getRoot(selectedPlayer)
    if not hum then return notify("no humanoid", THEME.danger) end
    pcall(function()
        hum.WalkSpeed = 0
        hum.JumpPower = 0
        hum.JumpHeight = 0
        if root then
            root.Anchored = true
        end
    end)
    notify("froze " .. selectedPlayer.Name, THEME.success)
end

local function doUnfreeze()
    if not selectedPlayer then return notify("no target", THEME.danger) end
    local hum = getHum(selectedPlayer)
    local root = getRoot(selectedPlayer)
    if not hum then return notify("no humanoid", THEME.danger) end
    pcall(function()
        hum.WalkSpeed = 16
        hum.JumpPower = 50
        hum.JumpHeight = 7.2
        if root then
            root.Anchored = false
        end
    end)
    notify("unfroze " .. selectedPlayer.Name, THEME.success)
end

local function doExplode()
    if not selectedPlayer then return notify("no target", THEME.danger) end
    local root = getRoot(selectedPlayer)
    if not root then return notify("no root", THEME.danger) end
    pcall(function()
        local e = Instance.new("Explosion")
        e.Position = root.Position
        e.BlastRadius = 12
        e.BlastPressure = 500000
        e.Parent = workspace
    end)
    notify("exploded " .. selectedPlayer.Name, THEME.success)
end

-- // TOGGLE
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == MENU_KEY then
        menuOpen = not menuOpen
        Main.Visible = menuOpen
    end
end)

-- // DRAG
local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)
TitleBar.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- // INIT
refreshPlayers()
Players.PlayerAdded:Connect(refreshPlayers)
Players.PlayerRemoving:Connect(function(p)
    if selectedPlayer == p then
        selectedPlayer = nil
        SelectedLbl.Text = "target: none"
        SelectedLbl.TextColor3 = THEME.dim
    end
    refreshPlayers()
end)
